# A1fp listwise/setwise experiment commands

## Repository setup

git clone https://github.com/ielab/llm-rankers.git external/llm-rankers

git -C external/llm-rankers rev-parse HEAD

# Optional environment setup before running baselines on a CUDA-capable machine with enough disk:
python3.10 -m venv .venv
. .venv/bin/activate
python -m pip install --upgrade pip
python -m pip install -e external/llm-rankers
python -m pip install torch transformers pyserini ir-datasets accelerate tiktoken sentencepiece

## Preflight / dry run

# Runs dependency, CUDA, BM25, model-cache, dataset-name, disk, and output-collision checks
# without launching generation. The full baseline runner executes this same preflight first.
CUDA_VISIBLE_DEVICES=0 bash experiments/a1fp_listwise_setwise/scripts/run_llm_rankers_baselines.sh --dry-run

# Optional: require the model config/tokenizer to already exist in the local HuggingFace cache.
REQUIRE_LOCAL_MODEL=1 CUDA_VISIBLE_DEVICES=0 bash experiments/a1fp_listwise_setwise/scripts/run_llm_rankers_baselines.sh --dry-run

## Commands executed locally

python experiments/a1fp_listwise_setwise/scripts/prepare_bm25.py

python experiments/a1fp_listwise_setwise/scripts/build_metrics.py

## BM25 generation fallback if repository BM25 files are absent

python -m pyserini.search.lucene \
  --threads 16 --batch-size 128 \
  --index msmarco-v1-passage \
  --topics dl19-passage \
  --output experiments/a1fp_listwise_setwise/runs/bm25.dl19.txt \
  --bm25 --k1 0.9 --b 0.4 --hits 100

python -m pyserini.search.lucene \
  --threads 16 --batch-size 128 \
  --index msmarco-v1-passage \
  --topics dl20-passage \
  --output experiments/a1fp_listwise_setwise/runs/bm25.dl20.txt \
  --bm25 --k1 0.9 --b 0.4 --hits 100

## BM25 sanity evaluation

python -m pyserini.eval.trec_eval -c -l 2 -m ndcg_cut.10 dl19-passage \
  experiments/a1fp_listwise_setwise/runs/bm25.dl19.txt

python -m pyserini.eval.trec_eval -c -l 2 -m ndcg_cut.10 dl20-passage \
  experiments/a1fp_listwise_setwise/runs/bm25.dl20.txt

## Listwise baseline: RankGPT-style listwise permutation generation

CUDA_VISIBLE_DEVICES=0 bash experiments/a1fp_listwise_setwise/scripts/run_llm_rankers_baselines.sh

## Individual baseline commands equivalent to the runner above

CUDA_VISIBLE_DEVICES=0 python external/llm-rankers/run.py \
  run \
    --model_name_or_path google/flan-t5-xl \
    --tokenizer_name_or_path google/flan-t5-xl \
    --run_path experiments/a1fp_listwise_setwise/runs/bm25.dl19.txt \
    --save_path experiments/a1fp_listwise_setwise/runs/listwise.rankgpt.flant5xl.w4s2.r1.dl19.txt \
    --ir_dataset_name msmarco-passage/trec-dl-2019 \
    --hits 100 \
    --query_length 32 \
    --passage_length 100 \
    --scoring generation \
    --device cuda \
  listwise \
    --window_size 4 \
    --step_size 2 \
    --num_repeat 1 \
  2>&1 | tee experiments/a1fp_listwise_setwise/logs/listwise.rankgpt.flant5xl.w4s2.r1.dl19.log

## Setwise baseline: Zhuang et al. setwise.heapsort

    --model_name_or_path google/flan-t5-xl \

    --save_path experiments/a1fp_listwise_setwise/runs/setwise.heapsort.flant5xl.c3.dl19.txt \
    --ir_dataset_name msmarco-passage/trec-dl-2019 \

    --query_length 32 \
    --passage_length 128 \
    --scoring generation \
    --device cuda \

    --num_child 2 \
    --method heapsort \
    --k 10 \
